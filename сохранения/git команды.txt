📘 Git — шпаргалка команд
🔧 Базовые команды


git init	                            Создать новый локальный репозиторий в текущей папке
git clone <url>	                        Скопировать удалённый репозиторий на локальную машину
git status	                            Показать состояние файлов (изменённые, новые, готовые к коммиту)
git add <файл>	                        Добавить файл в индекс (staging area) для следующего коммита
git add .	                            Добавить все изменённые файлы в индекс
git commit -m "сообщение"	            Создать коммит с добавленными файлами
git log	                                Показать историю коммитов
git log --oneline	                    Показать историю компактно (одна строка на коммит)

🌿 Ветки (branches)
git branch	                            Показать список локальных веток
git branch <имя>	                    Создать новую ветку
git checkout <ветка>	                Переключиться на другую ветку
git checkout -b <ветка>	                Создать и сразу переключиться на новую ветку
git switch <ветка>	                    Современный способ переключения веток
git switch -c <ветка>	                Создать и переключиться
git merge <ветка>	                    Влить указанную ветку в текущую
git branch -d <ветка>	                Удалить ветку (если она уже слита)
git branch -D <ветка>	                Принудительно удалить ветку (даже если не слита)


🔄 Отмена изменений и работа с коммитами
git restore <файл>	                    Отменить изменения в файле (незакоммиченные)
git restore --staged <файл>	            Убрать файл из индекса (но изменения оставить)
git checkout -- <файл>	                Старый способ отмены изменений в файле
git reset HEAD~1	                    Откатить последний коммит, изменения остаются в рабочей папке
git reset --hard HEAD~1	                Откатить последний коммит и удалить изменения
git revert <хеш>	                    Создать новый коммит, отменяющий указанный (безопасно для истории)
git commit --amend -m "сообщение"	    Изменить последний коммит (новое сообщение и/или добавить забытые файлы)
git clean -fd	                        Удалить все неотслеживаемые файлы и папки

☁️ Работа с удалённым репозиторием
git remote -v	                       Показать список удалённых репозиториев
git remote add origin <url>	           Добавить удалённый репозиторий с именем origin
git push origin <ветка>	               Отправить коммиты в удалённый репозиторий
git push -u origin <ветка>	           Отправить и запомнить связь с удалённой веткой
git push --force	                   Принудительно отправить изменения (перезаписать историю) — осторожно!
git pull	                           Забрать изменения с удалённого репозитория и сразу слить (fetch + merge)
git fetch	                           Забрать изменения, но не сливать
git clone <url>	                       Скопировать удалённый репозиторий

🧠 Полезные команды
git stash	                           Временно сохранить незакоммиченные изменения и очистить рабочую папку
git stash pop	                       Вернуть последние сохранённые изменения из stash
git stash list	                       Показать список сохранённых stash
git diff	                           Показать изменения в файлах (не в индексе)
git diff --staged	                   Показать изменения в индексе (что пойдёт в коммит)
git show <хеш>	                       Показать информацию о конкретном коммите
git blame <файл>	                   Показать, кто и когда менял каждую строку файла
git reflog	                           Показать историю действий (можно вернуть удалённые коммиты)
git log --graph --oneline --all        Показать историю коммитов в виде графа (дерева) с ветками
