📘 Docker — шпаргалка команд
🔧 Основные команды
docker version	                                  Показать версию Docker
docker info	                                      Информация о системе Docker
docker --help	                                  Помощь по любой команде

📦 Работа с образами (images)
docker images	                                  Список локальных образов
docker pull <образ>:<тег>	                      Скачать образ из реестра (Docker Hub)
docker push <образ>:<тег>	                      Загрузить образ в реестр
docker build -t <имя>:<тег> <путь>	              Собрать образ из Dockerfile
docker rmi <образ>	Удалить образ
docker tag <образ> <новое_имя>:<тег>	          Присвоить тег образу
docker history <образ>	                          Показать историю слоёв образа
docker save -o <файл.tar> <образ>	              Сохранить образ в tar-архив
docker load -i <файл.tar>	                      Загрузить образ из tar-архива

🚢 Работа с контейнерами
docker run <образ>	                              Запустить контейнер из образа
docker run -d <образ>	                          Запустить в фоне (detach)
docker run -it <образ> /bin/bash	              Запустить в интерактивном режиме с shell
docker run --name <имя> <образ>	                  Задать имя контейнера
docker run -p 8080:80 <образ>	                  Пробросить порт (хост:контейнер)
docker run -v /хост:/контейнер <образ>	          Пробросить папку (volume)
docker run --rm <образ>	                          Автоматически удалить контейнер после остановки
docker ps	                                      Список запущенных контейнеров
docker ps -a	                                  Список всех контейнеров (включая остановленные)
docker stop <контейнер>	                          Остановить контейнер
docker start <контейнер>	                      Запустить остановленный контейнер
docker restart <контейнер>	                      Перезапустить контейнер
docker pause <контейнер>	                      Приостановить контейнер
docker unpause <контейнер>	                      Возобновить контейнер
docker rm <контейнер>	                          Удалить контейнер
docker rm -f <контейнер>	                      Принудительно удалить работающий контейнер
docker container prune	                          Удалить все остановленные контейнеры

📊 Логи и отладка
docker logs <контейнер>	                          Показать логи контейнера
docker logs -f <контейнер>	                      Следить за логами (tail -f)
docker exec -it <контейнер> /bin/bash	          Зайти внутрь работающего контейнера
docker exec <контейнер> <команда>	              Выполнить команду в контейнере
docker cp <контейнер>:/путь/файл /хост/путь	      Скопировать файл из контейнера на хост
docker cp /хост/путь <контейнер>:/путь/файл	      Скопировать файл с хоста в контейнер
docker top <контейнер>	                          Показать процессы в контейнере
docker stats	                                  Статистика использования ресурсов контейнерами
docker inspect <контейнер/образ>	              Подробная информация в JSON

🧹 Очистка системы
docker system df	                              Показать использование диска Docker
docker system prune	                              Удалить всё неиспользуемое (контейнеры, сети, образы без тегов)
docker system prune -a	                          Удалить всё неиспользуемое, включая все остановленные контейнеры и неиспользуемые образы
docker image prune	                              Удалить неиспользуемые образы
docker container prune	                          Удалить все остановленные контейнеры
docker volume prune	                              Удалить неиспользуемые тома
docker network prune	                          Удалить неиспользуемые сети

📁 Тома (volumes)
docker volume create <имя>	                      Создать том
docker volume ls	                              Список томов
docker volume inspect <том>	                      Информация о томе
docker volume rm <том>	                          Удалить том
docker volume prune	                              Удалить неиспользуемые тома
docker run -v <том>:<путь> <образ>	              Подключить том к контейнеру
docker run --mount source=<том>,target=<путь> <образ>	  Современный синтаксис монтирования тома

🌐 Сети (networks)
docker network ls	                              Список сетей
docker network create <имя>	                      Создать сеть
docker network inspect <сеть>	                  Информация о сети
docker network connect <сеть> <контейнер>	      Подключить контейнер к сети
docker network disconnect <сеть> <контейнер>	  Отключить контейнер от сети
docker network rm <сеть>	                      Удалить сеть
docker network prune	                          Удалить неиспользуемые сети
docker run --network <сеть> <образ>	              Запустить контейнер в определённой сети

Типы сетей:
bridge — изолированная сеть на хосте (по умолчанию)
host — контейнер использует сеть хоста
none — без сети
overlay — для кластера Docker Swarm

🐙 Docker Compose
docker-compose up	                             Запустить все сервисы из docker-compose.yml
docker-compose up -d	                         Запустить в фоне
docker-compose down	                             Остановить и удалить контейнеры, сети
docker-compose down -v	                         Остановить и удалить контейнеры, сети и тома
docker-compose logs	                             Показать логи всех сервисов
docker-compose logs -f	                         Следить за логами
docker-compose ps	                             Список контейнеров проекта
docker-compose exec <сервис> <команда>	         Выполнить команду в работающем контейнере сервиса
docker-compose build	                         Пересобрать образы
docker-compose pull	                             Скачать образы для всех сервисов
docker-compose restart	                         Перезапустить все сервисы
docker-compose stop	                             Остановить все сервисы
docker-compose start	                         Запустить остановленные сервисы

📄 Dockerfile инструкции
FROM <образ>	                                 Базовый образ
WORKDIR <путь>	                                 Установить рабочую директорию
COPY <источник> <назначение>	                 Скопировать файлы в образ
ADD <источник> <назначение>	                     Копировать с поддержкой URL и tar-распаковки
RUN <команда>	                                 Выполнить команду при сборке образа
CMD ["команда", "арг1"]	                         Команда по умолчанию при запуске контейнера (можно переопределить)
ENTRYPOINT ["команда"]	                         Основная команда (не переопределяется)
EXPOSE <порт>	                                 Документировать порт, который слушает контейнер
ENV <ключ>=<значение>	                         Установить переменную окружения
ARG <имя>	                                     Переменная для сборки (не сохраняется в образе)
USER <пользователь>	                             Указать пользователя для запуска
VOLUME ["/путь"]	                             Создать точку монтирования для тома
LABEL <ключ>=<значение>	                         Добавить метаданные
HEALTHCHECK	                                     Проверка здоровья контейнера

🔍 Полезные комбинации
docker ps -a --format "table {{.Names}}\t{{.Status}}"	Красивый вывод с именами и статусом
docker rm -f $(docker ps -aq)	                        Удалить все контейнеры
docker rmi -f $(docker images -q)	                    Удалить все образы
docker stop $(docker ps -q)	                            Остановить все запущенные контейнеры
docker system df -v	                                    Детальный отчёт по использованию диска
