// Вам даны два непустых связанных списка, представляющих два неотрицательных целых числа.
// Цифры хранятся в обратном порядке, и каждый из их узлов содержит одну цифру. Сложите два числа и
// верните сумму в виде связанного списка.

// Вы можете предположить, что эти два числа не содержат никакого начального нуля, за исключением
// самого числа 0.

// [2 4 3]
// [5 6 4]
// [7 0 8]

//   [9,9,9,9,9,9,9]
//   [9,9,9,9]

//   [8,9,9,9,0,0,0,1] => 10 009 998

public class ListNode {

  public static void main(String[] args) {
    ListNode l1 = new ListNode(2);
    l1.next = new ListNode(4);
    l1.next.next = new ListNode(3);

    ListNode l2 = new ListNode(5);
    l2.next = new ListNode(6);
    l2.next.next = new ListNode(4);

    ListNode result = Solution.addTwoNumbers(l1, l2);
    printList(result);
  }

  public static void printList(ListNode head) {
    ListNode current = head;
    while (current != null) {
      System.out.print(current.val);
      if (current.next != null) System.out.print(" ");
      current = current.next;
    }
    System.out.println();
  }

  int val;
  ListNode next;

  ListNode() {}

  ListNode(int val) {
    this.val = val;
  }

  ListNode(int val, ListNode next) {
    this.val = val;
    this.next = next;
  }

  // Вложенный класс Solution
  class Solution {
    public static ListNode addTwoNumbers(ListNode l1, ListNode l2) {
      // Создаём фиктивный узел для удобства построения результата
      ListNode dummy = new ListNode(0);
      ListNode current = dummy; // указатель на текущую позицию в результате

      int carry = 0; // перенос при сложении (например, 8 + 7 = 15 → записываем 5, переносим 1)

      // Идём по обоим спискам, пока есть цифры или остался перенос
      while (l1 != null || l2 != null || carry != 0) {
        int val1 = (l1 != null) ? l1.val : 0; // если l1 закончился, считаем цифру за 0
        int val2 = (l2 != null) ? l2.val : 0; // аналогично для l2

        int sum = val1 + val2 + carry; // сумма цифр и переноса
        carry = sum / 10; // новое значение переноса (1 если >=10, иначе 0)
        int digit = sum % 10; // цифра, которую записываем в результат

        // Создаём новый узел с этой цифрой и добавляем в результат
        current.next = new ListNode(digit);
        current = current.next; // переходим к следующему узлу

        // Переходим к следующим цифрам в списках
        if (l1 != null) l1 = l1.next;
        if (l2 != null) l2 = l2.next;
      }

      // Возвращаем первый реальный узел (после фиктивного)
      return dummy.next;
    }
  }
}
